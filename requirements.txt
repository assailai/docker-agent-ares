# =============================================================================
# Ares Docker Agent - Python Dependencies (headless, zero-touch agent)
# =============================================================================
# The agent is an outbound HTTPS control-plane client plus a WebSocket data-plane
# tunnel. No web framework, no gRPC, no WireGuard, no local database.

# HTTP client (HTTP/2 + TLS 1.3) for the control plane.
httpx[http2]==0.28.1
# Public CA roots. Transitive via httpx, but agent.tlsconf imports it directly to union certifi's
# roots with the host store, so it is pinned here in its own right. Unlike the other pins this one
# ages badly (roots get added and distrusted), so move it forward when cutting a release.
# 2026.10.18 removed the emSign roots whose intermediate keys were compromised
# (GHSA-6q48-6c7w-9gw2); everything before it lets a certificate chained to them pass for
# ares.assailai.com. The updater pins the same version in requirements-updater.txt, and
# tests/test_image_pins.py keeps them equal.
certifi==2026.10.19
# WebSocket client for the data-plane tunnel.
websockets==14.1

# Config + env loading.
python-dotenv==1.0.1
pydantic==2.10.3
pydantic-settings==2.6.1

# Ed25519 signature verification for Network Assessment jobs (agent/assessment.py): the agent
# verifies the control plane's signature over a job before reading any target field, and signs its
# own evidence back. Byte-compatible with ares/domain/assessment on the control plane. The builder
# stage already has gcc + openssl-dev, so the wheel/build is available; floor matches the control
# plane's CVE floor (ares-v2 pins cryptography>=50.0.0).
cryptography==50.0.2

# SSH client for credentialed read-only config collection (Network Assessment COLLECT). Pinned at
# the latest to clear the image scan: asyncssh < 2.23.1 trips grype --fail-on high
# (GHSA-2wxc-x7rj-hg8f), and 2.24.x also clears GHSA-rw4j-r22c-9gc3. API used (connect,
# import_private_key, run) is stable across these minors.
asyncssh==2.24.1

# PostgreSQL client for the credentialed DATABASE collect tier (agent/collectors/db_query.py): a
# read-only catalog query (e.g. the installed-extension list) under a reader login from the local
# vault. Deliberately pg8000, the PURE-PYTHON driver (no C extension), so the Alpine image needs no
# extra build toolchain and the wheel is platform-independent. Transitive: scramp (SCRAM auth) and
# asn1crypto, both pure-Python.
pg8000==1.31.6

# LDAP client for the credentialed IDENTITY collect tier (agent/collectors/ldap_query.py): a
# read-only LDAPS bind + scoped search of an Active Directory domain controller under a reader login
# from the local vault. Deliberately ldap3, the PURE-PYTHON client (depends only on pyasn1, also
# pure-Python), so the Alpine image needs no OpenLDAP C libraries, unlike python-ldap.
ldap3==2.9.1

# vSphere client for the credentialed VIRTUALIZATION collect tier (agent/collectors/vsphere_collect.py):
# a read-only SOAP login + property read (e.g. ESXi host lockdown mode) of a vCenter/ESXi endpoint
# under a reader login from the local vault. pyvmomi is PURE-PYTHON; the base package has no required
# deps (the cryptography/lxml it names are only the optional `sso` extra, which simple user/pass auth
# does not use), so the Alpine image needs no extra build toolchain.
pyvmomi==9.1.2.0

# MySQL/MariaDB client for the credentialed DATABASE query tier (agent/collectors/mysql_query.py): a
# read-only catalog query (the account list) under a reader login from the local vault. Deliberately
# PyMySQL, the PURE-PYTHON driver (no C extension, no required deps for password auth), so the Alpine
# image needs no build toolchain, unlike mysqlclient.
pymysql==1.1.1
