# (GHSA-6q48-6c7w-9gw2); everything before it lets a certificate chained to them pass for
# ares.assailai.com. The updater pins the same version in requirements-updater.txt, and
# tests/test_image_pins.py keeps them equal.
certifi==2026.10.19
